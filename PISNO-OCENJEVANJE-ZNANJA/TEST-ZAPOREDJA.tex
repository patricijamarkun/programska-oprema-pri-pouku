\section*{Naloge}

\noindent\textit{Pri vsaki nalogi zapiši postopek reševanja.}

\begin{enumerate}
	\item \textbf{(6 točk)} Aritmetično zaporedje ima $a_3=11$ in $a_8=31$.
	\begin{enumerate}
		\item Izračunaj diferenco $d$ in prvi člen $a_1$ \textbf{(3 točke)}.
		\item Izračunaj $a_{25}$ \textbf{(1 točka)}.
		\item Izračunaj vsoto prvih 25 členov \textbf{(2 točki)}.
	\end{enumerate}
	\newpage

	\item \textbf{(6 točk)} Geometrijsko zaporedje ima $a_2=6$, $a_5=162$ in količnik $q>0$.
	\begin{enumerate}
		\item Izračunaj količnik $q$ in prvi člen $a_1$ \textbf{(3 točke)}.
		\item Izračunaj $a_{12}$ \textbf{(1 točka)}.
		\item Izračunaj vsoto prvih šestih členov \textbf{(2 točki)}.
	\end{enumerate}
	\newpage

	\item \textbf{(8 točk)} Zaporedje je podano s splošnim členom
	\[
		a_n=\frac{2n+1}{n+2}, \qquad n=1,2,\ldots
	\]
	\begin{enumerate}
		\item Zapiši prve štiri člene zaporedja \textbf{(2 točki)}.
		\item Ugotovi, ali je zaporedje naraščajoče, in utemelji odgovor. Določi tudi njegovo zgornjo mejo \textbf{(4 točke)}.
		\item Izračunaj limito zaporedja \textbf{(2 točki)}.
	\end{enumerate}
	\newpage

	\item \textbf{(8 točk)} Zaporedje je določeno rekurzivno:
	\[
		a_1=4, \qquad a_{n+1}=a_n+2n+1.
	\]
	\begin{enumerate}
		\item Izračunaj prvih pet členov zaporedja \textbf{(2 točki)}.
		\item Zapiši domnevo za splošni člen $a_n$ \textbf{(2 točki)}.
		\item Domnevo dokaži z matematično indukcijo \textbf{(4 točke)}.
	\end{enumerate}
	\newpage

	\item \textbf{(6 točk)} V gledališču je v prvi vrsti 18 sedežev, v vsaki naslednji vrsti pa so trije sedeži več kot v prejšnji. Gledališče ima 20 vrst.
	\begin{enumerate}
		\item Koliko sedežev je v zadnji vrsti? \textbf{(3 točke)}
		\item Koliko sedežev je v gledališču skupaj? \textbf{(3 točke)}
	\end{enumerate}
\end{enumerate}
